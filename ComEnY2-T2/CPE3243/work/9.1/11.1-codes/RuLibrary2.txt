DROP TABLE IF EXISTS `book`;

CREATE TABLE `book` (
  `id` INT NOT NULL AUTO_INCREMENT,
  `bookId5Digit` varchar(7) DEFAULT '00000',
  `bookId6Digit` varchar(8) DEFAULT '000000',
  `bookLanguage` varchar(4) DEFAULT '0',
  `bookTitle` varchar(300) DEFAULT NULL,
  `bookAuthor` varchar(200) DEFAULT NULL,
  `bookPublisher` varchar(200) DEFAULT NULL,
  `bookGroupNumber` varchar(30) DEFAULT NULL,
  `bookISBN10Digit` varchar(15) DEFAULT '0000000000',
  `bookISBN13Digit` varchar(20) DEFAULT '0000000000000',
  `bookBarcode` varchar(20) DEFAULT NULL,
  `bookMajor` varchar(10) DEFAULT NULL,
  `bookStatus` varchar(5) DEFAULT NULL,
  PRIMARY KEY (`id`)
) ENGINE = InnoDB DEFAULT CHARACTER SET = utf8mb4 COLLATE = utf8mb4_bin;
